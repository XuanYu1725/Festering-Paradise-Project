# 物品构造器：

物品构造器将物品数据构造出display

## 基本构造器

    item_builder:[]
    {
        id: 字符串，该物品的内部id
        reforge: 字符串，该物品的重铸id，可能不存在或为"default"
        template: 字符串 表明物品的模板类型 如"sword","hammer"
        rarity:浮点数，表示稀有度,
        variety:浮点数，可能不存在，表示该物品的变种
        bonus:{  加成数值
            damage:浮点数，表示基础攻击加成的百分数，如120表示x1.2倍或x120%,
            . . . #state中的属性，基础属性加成的百分数
        }
        <自定义构造器>
    }

## 自定义构造器：

    food_builder:{
        health:浮点数，表示血量回复,
        food:浮点数，表示饥饿回复,
        mana:浮点数，表示魔力回复,
        elf:浮点数，表示[elf]回复,
        applied_effects:[  状态效果
            {id:浮点数,Name:字符串,lvl:浮点数,等级,duration:浮点数,时长}
        ],
        using_events:[  自定义事件，在食用时触发
            {id:浮点数,Name:字符串}
        ]
    }

---

    skill_builder:{
        skills:[
            {id:浮点数,display:{Name:字符串,Lore:字符串，这俩都是原始json文本，但是需要过一遍字符串解析}}
        ],
        rightclick_events:[自定义事件，右键后触发
            {id:浮点数,Name:字符串,reclamation:浮点数,0或1,表示是否自动移除}
        ],
    }

---

    weapon_builder:{
        physical_damage: 若为浮点数，代表物理伤害
        若为复合标签，则结构为
        physical_damage:{
            puncture: 浮点数，穿刺伤害
            blunt: 浮点数，钝击伤害
            cutting: 浮点数，切割伤害
            display: 浮点数，展示的物理伤害，留空则展示三种伤害的最大值
        },
        magical_damage: 若为浮点数，代表魔法伤害
        若为复合标签，则结构为
        magical_damage:{
            fotia: 浮点数，火焰伤害
            pagos: 浮点数，冰冻伤害
            toxon: 浮点数，毒蚀伤害
            display: 浮点数，展示的魔法伤害，留空则展示三种伤害的最大值
        }

    }

---

    armor_builder:{
        physical_defence: 若为浮点数，代表物理防御
        若为复合标签，则结构为
        physical_defence:{
            puncture: 浮点数，穿刺防御
            blunt: 浮点数，钝击防御
            cutting: 浮点数，切割防御
            display: 浮点数，展示的物理防御，留空则展示三种防御的最大值
        },
        magical_defence: 若为浮点数，代表魔法防御
        若为复合标签，则结构为
        magical_defence:{
            fotia: 浮点数，火焰防御
            pagos: 浮点数，冰冻防御
            toxon: 浮点数，毒蚀防御
            display: 浮点数，展示的魔法防御，留空则展示三种防御的最大值
        }
        hit_events:[自定义事件，受伤后触发，装备特性通过这个实现
            {id:浮点数,Name:字符串,reclamation:浮点数,0或1,表示是否自动移除}
        ]
    }

---

    container_builder:{

        }
        items:{
            ... 内容物 符合item_stacks
        }  
    }
